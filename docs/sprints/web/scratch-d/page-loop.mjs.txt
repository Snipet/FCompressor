import { createRequire } from 'node:module';
import { createHash } from 'node:crypto';
const { fcmpSynthLoop } = createRequire(import.meta.url)(process.argv[2]);
const db = (v) => (20 * Math.log10(v)).toFixed(2);
for (const rate of [48000, 44100, 96000]) {
  const t0 = performance.now();
  const loop = fcmpSynthLoop(rate);
  const ms = performance.now() - t0;
  const { left: L, right: R, frames: n } = loop;
  let peak = 0, sum = 0, dc = 0, finite = true, maxStep = 0, maxStepAt = 0;
  for (let i = 0; i < n; i += 1) {
    peak = Math.max(peak, Math.abs(L[i]), Math.abs(R[i]));
    sum += L[i] * L[i] + R[i] * R[i]; dc += L[i] + R[i];
    finite = finite && Number.isFinite(L[i]) && Number.isFinite(R[i]);
    const step = Math.abs(L[(i + 1) % n] - L[i]);
    if (step > maxStep) { maxStep = step; maxStepAt = i; }
  }
  const rms = Math.sqrt(sum / (2 * n));
  const seam = Math.abs(L[0] - L[n - 1]), seamR = Math.abs(R[0] - R[n - 1]);
  // short-term RMS (50 ms windows): the range a compressor will ride
  const w = Math.round(0.05 * rate); let lo = 1e9, hi = 0;
  for (let a = 0; a + w <= n; a += w) { let s = 0; for (let i = a; i < a + w; i += 1) s += L[i] * L[i]; s = Math.sqrt(s / w); lo = Math.min(lo, s); hi = Math.max(hi, s); }
  const h = createHash('sha256').update(Buffer.from(L.buffer)).update(Buffer.from(R.buffer)).digest('hex').slice(0, 16);
  console.log(`rate ${rate}: ${n} frames (${(n / rate).toFixed(3)} s) in ${ms.toFixed(0)} ms; finite ${finite}; peak ${db(peak)} dBFS; RMS ${db(rms)} dBFS; crest ${(db(peak) - db(rms)).toFixed(1)} dB; DC ${(dc / (2 * n)).toExponential(2)}`);
  console.log(`   seam step L ${seam.toExponential(2)} R ${seamR.toExponential(2)}; largest step anywhere ${maxStep.toFixed(4)} at ${(maxStepAt / rate).toFixed(4)} s; 50 ms RMS range ${db(lo)} .. ${db(hi)} dBFS; sha ${h}`);
}
