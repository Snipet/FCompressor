#!/usr/bin/env node
// SCOUT SCRATCH: the page's life in headless Chrome, step by step.  node scenario.mjs <dir> [chrome flags...]
import { spawn } from 'node:child_process';
import { mkdtempSync, rmSync, writeFileSync } from 'node:fs';
import { join, resolve } from 'node:path';
const dir = resolve(process.argv[2]); const flags = process.argv.slice(3);
const chrome = '/Applications/Google Chrome.app/Contents/MacOS/Google Chrome';
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
const children = []; let profile = '';
const finish = (code) => { for (const c of children) { try { c.kill('SIGKILL'); } catch {} } if (profile) { try { rmSync(profile, { recursive: true, force: true, maxRetries: 5 }); } catch {} } process.exit(code); };
const lineFrom = (child, pattern) => new Promise((done) => { let seen = ''; const look = (c) => { seen += c; const m = seen.match(pattern); if (m) done(m); }; child.stdout.on('data', look); child.stderr.on('data', look); child.on('exit', () => done(null)); setTimeout(() => done(null), 20000); });
const server = spawn('python3', ['-u', '-m', 'http.server', '0', '--bind', '127.0.0.1', '--directory', dir], { stdio: ['ignore', 'pipe', 'pipe'] });
children.push(server);
const serving = await lineFrom(server, /Serving HTTP on 127\.0\.0\.1 port (\d+)/);
const base = `http://127.0.0.1:${serving[1]}/`;
profile = mkdtempSync(join(dir, '..', 'profile-'));
const browser = spawn(chrome, ['--headless=new', '--remote-debugging-port=0', `--user-data-dir=${profile}`, '--no-first-run', '--no-default-browser-check', '--disable-extensions', '--use-angle=metal', '--mute-audio', '--window-size=1200,900', ...flags, 'about:blank'], { stdio: ['ignore', 'pipe', 'pipe'] });
children.push(browser);
const listening = await lineFrom(browser, /DevTools listening on (ws:\/\/[^\s]+)/);
const socket = new WebSocket(listening[1]);
await new Promise((open) => socket.addEventListener('open', open, { once: true }));
let nextId = 0; const waiting = new Map(); const consoleLines = [];
socket.addEventListener('message', (event) => {
  const m = JSON.parse(event.data);
  if (m.id !== undefined && waiting.has(m.id)) { const w = waiting.get(m.id); waiting.delete(m.id); m.error ? w.reject(new Error(m.error.message)) : w.answer(m.result); }
  else if (m.method === 'Runtime.exceptionThrown') consoleLines.push('EXCEPTION ' + JSON.stringify(m.params.exceptionDetails.exception && m.params.exceptionDetails.exception.description || m.params.exceptionDetails.text));
  else if (m.method === 'Log.entryAdded') consoleLines.push('LOG ' + m.params.entry.level + ' ' + m.params.entry.text);
  else if (m.method === 'Runtime.consoleAPICalled') consoleLines.push('CONSOLE ' + m.params.type + ' ' + m.params.args.map((a) => a.value || a.description).join(' '));
});
const send = (method, params = {}, sessionId) => new Promise((answer, reject) => { const id = ++nextId; waiting.set(id, { answer, reject }); socket.send(JSON.stringify({ id, method, params, sessionId })); });
const { targetId } = await send('Target.createTarget', { url: 'about:blank' });
const { sessionId } = await send('Target.attachToTarget', { targetId, flatten: true });
await send('Runtime.enable', {}, sessionId); await send('Log.enable', {}, sessionId); await send('Page.enable', {}, sessionId);
const ev = async (expression) => { const r = await send('Runtime.evaluate', { expression, returnByValue: true, awaitPromise: true }, sessionId); return r.exceptionDetails ? 'EXC ' + (r.exceptionDetails.exception && r.exceptionDetails.exception.description) : r.result.value; };
const click = async (x, y) => { for (const type of ['mousePressed', 'mouseReleased']) await send('Input.dispatchMouseEvent', { type, x, y, button: 'left', buttons: type === 'mousePressed' ? 1 : 0, clickCount: 1 }, sessionId); };
const shot = async (name) => { const { data } = await send('Page.captureScreenshot', { format: 'png' }, sessionId); writeFileSync(join(dir, '..', name), Buffer.from(data, 'base64')); };
const say = (k, v) => console.log(`${k}: ${typeof v === 'string' ? v : JSON.stringify(v)}`);
const info = "[0,1,2,3,4,5,6,7].map((i) => Module._fcmp_ui_info(i))";
const stats = "fcmpDebug.ask(fcmpDebug.node(), 'stats').then((s) => ({ quanta: s.quanta, inCh: s.inChannels, frames: s.lastFrames }))";
try {
  await send('Page.navigate', { url: base + 'index.html' }, sessionId);
  await sleep(800);
  say('idle: overlay text', await ev("document.getElementById('fcmp-status').textContent"));
  say('idle: AudioContext made before the click', await ev("fcmpDebug.ctx() === null ? 'none' : 'yes'"));
  const at = await ev("(() => { const r = document.getElementById('fcmp-start').getBoundingClientRect(); return [r.left + r.width / 2, r.top + r.height / 2]; })()");
  const t0 = Date.now();
  await click(at[0], at[1]);
  while (await ev('fcmpDebug.state()') !== 'running' && Date.now() - t0 < 15000) await sleep(20);
  say('click -> running in ms', Date.now() - t0);
  say('context', await ev("[fcmpDebug.ctx().state, fcmpDebug.ctx().sampleRate, fcmpDebug.ctx().baseLatency]"));
  await sleep(1000);
  say('info after 1 s [atlasHi, atlasLo, ok, frames, replies, flags, refused, bypass]', await ev(info));
  // capture the Params records the editor posts for: HQ, then LOOKAHEAD 20 MS, then STD and OFF again
  await ev("(() => { const p = Module.fcmp.port; const orig = p.postMessage.bind(p); window.recs = []; p.postMessage = (m, t) => { if (m.b && m.n === 140) window.recs.push(Array.from(new Uint8Array(m.b, 0, 140))); return orig(m, t); }; })()");
  const c = await ev("(() => { const r = document.getElementById('fcmp-canvas').getBoundingClientRect(); return [r.left, r.top, r.width, r.height]; })()");
  const q0 = await ev(stats);
  await click(c[0] + 531, c[1] + 70); await sleep(700);          // HQ
  say('after HQ: stats, info', await ev(`Promise.all([${stats}, ${info}])`));
  await click(c[0] + 529, c[1] + 96); await sleep(700);          // LOOKAHEAD 20 MS
  say('after 20 MS: stats, info', await ev(`Promise.all([${stats}, ${info}])`));
  await shot('shot-hq.png');
  const recs = await ev('window.recs');
  writeFileSync(join(dir, '..', 'records.json'), JSON.stringify(recs));
  say('records captured', recs.length);
  // memory and the long run
  say('JS heap', await ev('performance.memory ? Math.round(performance.memory.usedJSHeapSize / 1e6) + " MB" : "n/a"'));
} catch (e) { console.log('SCENARIO ERROR ' + (e && e.stack || e)); }
for (const l of consoleLines.slice(0, 20)) console.log(l);
finish(0);
