// Item 7: a QUALITY change (STD to HQ) and a LOOKAHEAD budget change take effect; audio continues; does it click.
// The output is recorded by a second AudioWorklet of the driver's own (a Blob module), tapped off the engine's node.
import { serve, chrome, sleep, Report, HERE, cleanUp, PID } from './lib.mjs';
const R = new Report();
const server = await serve(`${HERE}/site-hook`);
const b = await chrome({ width: 1280, height: 800 });
try {
  const p = await b.page(null);
  await p.metrics(1280, 800, 1);
  await p.go(`${server.base}/`, 1500);
  await p.start(); await sleep(1000); await p.tap(); await sleep(400);
  await p.ev(`(async () => {
    const src = 'class Rec extends AudioWorkletProcessor { constructor() { super(); this.n = 48000 * 6; this.l = new Float32Array(this.n); this.at = 0; this.total = 0; this.port.onmessage = () => { const out = new Float32Array(this.n); const k = this.at; out.set(this.l.subarray(k)); out.set(this.l.subarray(0, k), this.n - k); this.port.postMessage({ total: this.total, l: out }, [out.buffer]); }; } process(inputs) { const i = inputs[0]; if (i && i[0]) { const a = i[0]; for (let j = 0; j < a.length; j += 1) { this.l[this.at] = a[j]; this.at = (this.at + 1) % this.n; } this.total += a.length; } return true; } } registerProcessor("scout-rec", Rec);';
    const ctx = fcmpPage.context();
    await ctx.audioWorklet.addModule(URL.createObjectURL(new Blob([src], { type: 'text/javascript' })));
    const rec = new AudioWorkletNode(ctx, 'scout-rec', { numberOfInputs: 1, numberOfOutputs: 0, channelCount: 2, channelCountMode: 'explicit' });
    fcmpPage.node().connect(rec);
    globalThis.__rec = () => new Promise((done) => { rec.port.onmessage = (e) => done(e.data); rec.port.postMessage(0); });
    // the last seconds of the output, analysed: the largest sample-to-sample step, where, runs of exact zeros, and the step
    // sizes' high percentiles (what the programme itself does)
    globalThis.__analyse = async (seconds) => { const d = await __rec(); const x = d.l.subarray(d.l.length - Math.round(seconds * 48000));
      let max = 0, at = 0; const steps = new Float32Array(x.length - 1);
      for (let i = 1; i < x.length; i += 1) { const s = Math.abs(x[i] - x[i - 1]); steps[i - 1] = s; if (s > max) { max = s; at = i; } }
      const sorted = Float32Array.from(steps).sort();
      let zeroRun = 0, longest = 0, longestAt = 0; for (let i = 0; i < x.length; i += 1) { if (x[i] === 0) { zeroRun += 1; if (zeroRun > longest) { longest = zeroRun; longestAt = i - zeroRun + 1; } } else zeroRun = 0; }
      let peak = 0; for (const v of x) peak = Math.max(peak, Math.abs(v));
      return { total: d.total, frames: x.length, peak, maxStep: max, maxStepAtMsFromEnd: (x.length - at) / 48, around: Array.from(x.subarray(Math.max(0, at - 4), at + 4)).map((v) => +v.toFixed(4)),
               p999: sorted[Math.floor(sorted.length * 0.999)], p9999: sorted[Math.floor(sorted.length * 0.9999)], longestZeroRun: longest, zeroRunAtMsFromEnd: (x.length - longestAt) / 48 }; };
  })()`);
  await sleep(2500);
  const base = JSON.parse(await p.ev('__analyse(2).then(JSON.stringify)'));
  R.note('7.baseline', base);
  const look = async () => { const st = await p.status(); const ws = await p.stats(); const t = await p.tapRead(); return { statusLatency: st.latency, workletLatency: ws.latency, replyLatency: t.reply.latency, quanta: ws.quanta, refused: ws.refused, ok: ws.ok, quality: t.last ? t.last.v[PID.quality] : null, labudget: t.last ? t.last.v[PID.labudget] : null, records: ws.records }; };
  const change = async (id, group, cell, wantLatencyChange) => {
    const before = await look();
    const t0 = Date.now();
    await p.clickItem(cell, { parent: group, settle: 0 });
    await sleep(700);
    const an = JSON.parse(await p.ev('__analyse(1.2).then(JSON.stringify)'));
    const after = await look();
    await sleep(600);
    const later = await look();
    const settings = { quality: await p.value('Quality'), budget: await p.value('Lookahead budget') };
    const expectedQuanta = (later.quanta - after.quanta) / 0.6;
    const ratio = an.maxStep / base.p9999;
    R.row(after.ok && after.refused === 0 && later.quanta > after.quanta && (wantLatencyChange ? after.statusLatency !== before.statusLatency : true) && after.statusLatency === after.workletLatency && after.replyLatency === after.workletLatency,
          id, { before: { latency: before.statusLatency }, after: { latency: after.statusLatency, worklet: after.workletLatency, reply: after.replyLatency, quality: after.quality, labudget: after.labudget }, cells: settings,
                quantaPerSecondAfter: Math.round(expectedQuanta), refused: after.refused,
                click: { maxStep: +an.maxStep.toFixed(4), atMsBeforeEndOfWindow: +an.maxStepAtMsFromEnd.toFixed(1), samplesAround: an.around, programmeP9999: +base.p9999.toFixed(4), timesProgramme: +ratio.toFixed(1), longestZeroRunSamples: an.longestZeroRun, zeroRunMsBeforeEnd: +an.zeroRunAtMsFromEnd.toFixed(1), peak: +an.peak.toFixed(3) } });
    return { an, after };
  };
  await change('7.quality.std-to-hq', 'Quality', 'HQ', true);
  await sleep(1500);
  await change('7.lookahead.off-to-5ms', 'Lookahead budget', '5 ms', true);
  await sleep(1500);
  await change('7.lookahead.5-to-20ms', 'Lookahead budget', '20 ms', true);
  await sleep(1500);
  await change('7.quality.hq-to-eco', 'Quality', 'Eco', true);
  await sleep(1500);
  await change('7.quality.eco-to-std', 'Quality', 'Std', true);
  await sleep(1500);
  await change('7.lookahead.20-to-off', 'Lookahead budget', 'Off', true);
  await sleep(1500);
  // a control: an ordinary edit (THRESHOLD by a typed value) does not step like that
  const slot = await p.need('THRESHOLD');
  await p.click(slot.x + slot.w / 2, slot.y + 44); await p.type('-30'); await p.key('Enter'); await sleep(700);
  const edit = JSON.parse(await p.ev('__analyse(1.2).then(JSON.stringify)'));
  R.note('7.control.threshold-edit', { maxStep: +edit.maxStep.toFixed(4), timesProgramme: +(edit.maxStep / base.p9999).toFixed(1), longestZeroRun: edit.longestZeroRun });
  // the settings screen's cells do the same, and its LATENCY rows follow
  await p.clickItem('Settings'); await sleep(500);
  const lat0 = await p.value('Latency');
  await p.clickItem('HQ', { parent: 'Quality' }); await sleep(900);
  const lat1 = await p.value('Latency'); const st = await p.status();
  const os = await p.value('Oversampling');
  const shot = await p.shot('7-settings-hq.png');
  R.row(lat0 !== lat1 && lat1.startsWith(`${st.latency} SAMPLES`), '7.settings.latency-row-follows', { before: lat0, after: lat1, statusLatency: st.latency, oversampling: os, png: shot });
  await p.clickItem('Std', { parent: 'Quality' }); await sleep(600);
  await p.key('Escape');
  // the page's list of differences says it may click
  const says = await p.ev("Array.from(document.querySelectorAll('#fcmp-notes li')).map((l) => l.textContent.replace(/\\s+/g, ' ')).find((t) => /QUALITY OR LOOKAHEAD/.test(t)) || ''");
  R.row(/MAY\s+CLICK/.test(says), '7.page-says-may-click', says);
  R.row(p.consoleLines.length === 0, '7.console', p.consoleLines.length ? p.consoleLines : 'nothing on the console');
} catch (e) { R.row(false, 'driver', String(e && e.stack || e)); }
finally { b.kill(); server.kill(); cleanUp(); }
console.log(`\n${R.rows.filter((r) => r.ok === false).length} FAIL of ${R.rows.filter((r) => r.ok !== null).length}`);
process.exit(0);
