// Scratch test-only processor: runs the probe in its constructor, in a message handler and inside process().
const CASES = [
  // name, function, a, b: 1e-20f = 0x1e3ce508, 1e20f = 0x60ad78ec, 1e-40f (denormal) = 0x000116c2, FLT_MIN = 0x00800000
  ['ftz: 1e-20 * 1e-20 (f32.mul)', 'mul32', 0x1e3ce508, 0x1e3ce508],
  ['ftz: 1e-20 * 1e-20 (f32x4.mul)', 'mulv', 0x1e3ce508, 0x1e3ce508],
  ['ftz: FLT_MIN + -1.5*FLT_MIN... (f32.add FLT_MIN, -0x00c00000)', 'add32', 0x00800000, 0x80c00000],
  ['daz: denormal 1e-40 * 1e20 (f32.mul)', 'mul32', 0x000116c2, 0x60ad78ec],
  ['daz: denormal 1e-40 * 1e20 (f32x4.mul)', 'mulv', 0x000116c2, 0x60ad78ec],
  ['demote: (double)1e-20 * (double)1e-20 -> f32', 'demote', 0x1e3ce508, 0x1e3ce508],
  ['f64: 2^-600 * 2^-450 (denormal double)', 'mul64hi', 0x1a700000, 0x23d00000],
];
function probe(x) {
  const out = {};
  for (const [name, fn, a, b] of CASES) out[name] = (x[fn](a, b) >>> 0).toString(16).padStart(8, '0');
  // JavaScript itself: a denormal double.
  const d = new Float64Array([2.2250738585072014e-308, 0.5]);
  out['js: DBL_MIN * 0.5'] = String(d[0] * d[1]);
  return out;
}
class FpModeProcessor extends AudioWorkletProcessor {
  constructor(options) {
    super();
    this.x = new WebAssembly.Instance(new WebAssembly.Module(options.processorOptions.wasm), {}).exports;
    this.inProcess = null;
    this.inConstructor = probe(this.x);
    this.port.onmessage = () => this.port.postMessage({ inConstructor: this.inConstructor, inHandler: probe(this.x), inProcess: this.inProcess });
  }
  process() {
    if (this.inProcess === null) this.inProcess = probe(this.x);
    return true;
  }
}
registerProcessor('fcmp-fpmode', FpModeProcessor);
