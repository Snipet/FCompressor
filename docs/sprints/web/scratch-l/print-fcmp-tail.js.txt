// Scratch prototype (scout-l/audio): the cost of silence per Mode in this browser.
//   main    the engine on this thread, the silence gate off, as web.engine.tail (2 s of the print program, 10 s of zeros,
//           1/3 s windows, best of 3, the worst adjacent pair against the signal)
//   worklet the SHIPPED worklet in an OfflineAudioContext, timed by the render's wall clock: A = 12 s of the program;
//           B = 2 s of it then 10 s of zeros (the gate as shipped); C = 2 s then 10 s of a +-2^-120 floor (normal
//           floats at the denormal boundary: the gate stays open, every product below 1 is in the denormal range)
const QUANTUM = 128;
const say = (line) => { document.getElementById('funkgui-log').textContent += `${line}\n`; };
const within = (ms, what, promise) => { let t = 0; const late = new Promise((_, rej) => { t = setTimeout(() => rej(new Error(`${what}: no answer in ${ms} ms`)), ms); }); return Promise.race([promise, late]).finally(() => clearTimeout(t)); };
const makeNode = (ctx, wasm) => within(10000, 'the audio processor', new Promise((made, bad) => {
  const n = new AudioWorkletNode(ctx, 'fcmp-engine', { numberOfInputs: 1, numberOfOutputs: 1, outputChannelCount: [2], channelCount: 2, channelCountMode: 'explicit', channelInterpretation: 'speakers', processorOptions: { wasm } });
  const hear = (event) => { const m = event.data; if (!m || (m.fcmp !== 'ready' && m.fcmp !== 'error')) return; n.port.removeEventListener('message', hear); if (m.fcmp === 'ready') made(n); else bad(new Error(m.error)); };
  n.port.addEventListener('message', hear); n.port.start();
}));
const ask = (n, what) => within(10000, what, new Promise((answered) => { const hear = (event) => { if (!event.data || event.data.fcmp !== what) return; n.port.removeEventListener('message', hear); answered(event.data); }; n.port.addEventListener('message', hear); n.port.postMessage({ fcmp: what }); }));

async function main() {
  say(`NOTE     ${navigator.userAgent}`);
  const cfg = await (await fetch('fcmp-tail-config.json')).json();
  const engineBytes = await (await fetch('fcmp-engine.wasm')).arrayBuffer();
  const p = (await WebAssembly.instantiate(await (await fetch('fcmp-tail.wasm')).arrayBuffer(), {})).instance.exports;
  p._initialize();
  const frames = p.fcmp_print_frames();
  const pL = p.malloc(frames * 4), pR = p.malloc(frames * 4), pRec = p.malloc(140);
  const pu8 = new Uint8Array(p.memory.buffer), pf32 = new Float32Array(p.memory.buffer);
  p.fcmp_print_program(pL, pR);
  const progL = pf32.slice(pL / 4, pL / 4 + frames), progR = pf32.slice(pR / 4, pR / 4 + frames);
  const cstr = (at) => { let s = ''; for (let i = at; pu8[i] !== 0; i += 1) s += String.fromCharCode(pu8[i]); return s; };
  const record = (m, quality) => { p.fcmp_print_record(m, 0, quality, 0, -1, 1, pRec); return pu8.slice(pRec, pRec + 140); };
  const modes = [];
  for (let i = 0; i < p.fcmp_print_modes(); i += 1) modes.push(cstr(p.fcmp_print_key(i)));

  // ---- main thread, gate off -------------------------------------------------------------------------------------
  const x = (await WebAssembly.instantiate(engineBytes, {})).instance.exports;
  x._initialize();
  const f32 = new Float32Array(x.memory.buffer), u8 = new Uint8Array(x.memory.buffer);
  const scratch = x.malloc(4 * QUANTUM * 4 + 160);
  const [inL, inR, outL, outR] = [0, 1, 2, 3].map((i) => scratch + i * QUANTUM * 4);
  const msg = scratch + 4 * QUANTUM * 4;
  const WINDOW = 125, ACTIVE = 6, SILENT = 30;
  const run = (m, silentWindows) => {
    const e = x.fcmp_web_create();
    u8.set(record(m, 1), msg);
    x.fcmp_web_post(e, msg, 140);
    x.fcmp_web_configure(e, 48000, QUANTUM);
    x.fcmp_web_set_gate(e, 0);
    const active = [], silent = [];
    for (let w = 0; w < ACTIVE + silentWindows; w += 1) {
      if (w === ACTIVE) f32.fill(0, inL / 4, inL / 4 + 2 * QUANTUM);
      const from = performance.now();
      for (let q = 0; q < WINDOW; q += 1) {
        if (w < ACTIVE) { const at = (w * WINDOW + q) * QUANTUM; f32.set(progL.subarray(at, at + QUANTUM), inL / 4); f32.set(progR.subarray(at, at + QUANTUM), inR / 4); }
        x.fcmp_web_process(e, inL, inR, outL, outR, QUANTUM);
      }
      (w < ACTIVE ? active : silent).push((performance.now() - from) / WINDOW);
    }
    x.fcmp_web_destroy(e);
    active.sort((a, b) => a - b);
    return { active: active[(ACTIVE - 1) >> 1], silent };
  };
  const tMain = performance.now();
  for (let m = 0; m < modes.length; m += 1) {
    run(m, 3);
    let best = null;
    for (let i = 0; i < 3; i += 1) { const t = run(m, SILENT); best = best === null ? t : { active: Math.min(best.active, t.active), silent: best.silent.map((s, w) => Math.min(s, t.silent[w])) }; }
    let worst = 0, worstAt = 0;
    for (let w = 0; w + 1 < SILENT; w += 1) { const pair = Math.min(best.silent[w], best.silent[w + 1]); if (pair > worst) [worst, worstAt] = [pair, w]; }
    say(`${worst < 2 * best.active ? 'PASS' : 'FAIL'}     web.live.tail ${modes[m]} main.cost: signal ${(best.active * 1000).toFixed(1)} us a quantum, silence at worst ${(worst * 1000).toFixed(1)} us over 2/3 s from ${(worstAt / 3).toFixed(1)} s (x${(worst / best.active).toFixed(2)}, limit x2; gate off, best of 3)`);
  }
  say(`NOTE     main thread: ${modes.length} Modes in ${((performance.now() - tMain) / 1000).toFixed(1)} s`);
  if (!cfg.worklet) return;

  // ---- through the worklet ---------------------------------------------------------------------------------------
  const SECONDS = 12, total = SECONDS * 48000, burst = 2 * 48000;
  const floor = new Float32Array(96);
  for (let i = 0; i < 96; i += 1) floor[i] = (i < 48 ? 1 : -1) * 2 ** -120;
  const material = (kind) => {
    const l = new Float32Array(total), r = new Float32Array(total);
    for (let i = 0; i < total; i += 1) {
      if (kind === 'A' || i < burst) { l[i] = progL[i % frames]; r[i] = progR[i % frames]; }
      else if (kind === 'C') { l[i] = floor[i % 96]; r[i] = -floor[i % 96]; }
    }
    return [l, r];
  };
  const mats = { A: material('A'), B: material('B'), C: material('C') };
  const render = async (m, kind) => {
    const ctx = new OfflineAudioContext({ numberOfChannels: 2, length: total, sampleRate: 48000 });
    await ctx.audioWorklet.addModule('fcmp-worklet.js');
    const node = await makeNode(ctx, engineBytes);
    node.port.postMessage(record(m, 2).buffer);
    node.port.postMessage(record(m, 1).buffer);
    await ask(node, 'stats');
    const buffer = ctx.createBuffer(2, total, 48000);
    buffer.copyToChannel(mats[kind][0], 0); buffer.copyToChannel(mats[kind][1], 1);
    const source = ctx.createBufferSource(); source.buffer = buffer; source.connect(node).connect(ctx.destination); source.start();
    const t0 = performance.now();
    const out = await ctx.startRendering();
    const ms = performance.now() - t0;
    let peakTail = 0; const d = out.getChannelData(0);
    for (let i = total - 48000; i < total; i += 1) peakTail = Math.max(peakTail, Math.abs(d[i]));
    return { ms, peakTail };
  };
  const tW = performance.now();
  for (const key of cfg.modes === 'all' ? modes : cfg.modes) {
    const m = modes.indexOf(key);
    const best = { A: Infinity, B: Infinity, C: Infinity }; let tailC = 0, tailB = 0;
    for (let rep = 0; rep < cfg.reps; rep += 1)
      for (const kind of ['A', 'B', 'C']) { const r = await render(m, kind); best[kind] = Math.min(best[kind], r.ms); if (kind === 'C') tailC = r.peakTail; if (kind === 'B') tailB = r.peakTail; }
    say(`NOTE     web.live.tail ${key} worklet: signal ${best.A.toFixed(0)} ms for ${SECONDS} s (${(SECONDS * 1000 / best.A).toFixed(0)}x real time); burst + zeros ${best.B.toFixed(0)} ms (x${(best.B / best.A).toFixed(2)}, last-second peak ${tailB}); burst + 2^-120 floor ${best.C.toFixed(0)} ms (x${(best.C / best.A).toFixed(2)}, last-second peak ${tailC.toExponential(2)})`);
  }
  say(`NOTE     worklet: ${((performance.now() - tW) / 1000).toFixed(1)} s`);
}
main().catch((e) => say(`FAIL     ${e && e.stack || e}`)).then(() => { document.title = document.getElementById('funkgui-log').textContent.includes('FAIL ') ? 'FAIL: tail' : 'PASS'; });
